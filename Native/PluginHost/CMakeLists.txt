# mfplughost: the music editor's out-of-process helper (docs/design/natives.md, ADR 0146) — ONE executable:
#   Windows: mfplughost.exe   Linux / macOS: mfplughost
# Editor-only: shipped from MainframeEngine.Editor/runtimes/<rid>/native/, never with games.
#   mfplughost --encode <in.wav> <out.ogg> --quality <0..10>   WAV (16-bit PCM / 32-bit float) -> Ogg Vorbis
#   mfplughost --serve <socket-path>                            the editor's helper (length-prefixed binary protocol)
#   mfplughost --scan <bundle.vst3>                             one bundle's VST3 classes as JSON (one process per bundle)
#   mfplughost --version
# Third-party code is static: libogg 1.3.6 and libvorbis/libvorbisenc 1.3.7 (BSD-3-Clause), the VST 3 SDK 3.8.1 (MIT;
# base, pluginterfaces, public.sdk hosting — no VSTGUI), RtMidi 6.0.0 (MIT; CoreMIDI / WinMM / ALSA), submodules in
# external/. The test plugin bundle (tests/plugins, CTest fixture) is never shipped.
cmake_minimum_required(VERSION 3.25)

project(mfplughost VERSION 0.1.0 LANGUAGES C CXX)

if(PROJECT_IS_TOP_LEVEL)
	include(CTest)
endif()

option(MF_NATIVES_WERROR "Treat warnings in our own (non third-party) native code as errors" ON)
option(MF_PLUGINHOST_VST3 "Host VST3 plugins (VST 3 SDK 3.8, external/vst3sdk)" ON)
option(MF_PLUGINHOST_MIDI "MIDI input devices (RtMidi 6, external/rtmidi)" ON)

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>") # static CRT (/MT)

set(OGG "${CMAKE_CURRENT_SOURCE_DIR}/external/ogg")
set(VORBIS "${CMAKE_CURRENT_SOURCE_DIR}/external/vorbis")

# libogg's config_types.h is normally generated by configure/CMake; every supported compiler has <stdint.h>.
set(MF_OGG_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${MF_OGG_GENERATED}/ogg/config_types.h.tmp" [=[
#ifndef __CONFIG_TYPES_H__
#define __CONFIG_TYPES_H__
#include <stdint.h>
typedef int16_t ogg_int16_t;
typedef uint16_t ogg_uint16_t;
typedef int32_t ogg_int32_t;
typedef uint32_t ogg_uint32_t;
typedef int64_t ogg_int64_t;
typedef uint64_t ogg_uint64_t;
#endif
]=])
configure_file("${MF_OGG_GENERATED}/ogg/config_types.h.tmp" "${MF_OGG_GENERATED}/ogg/config_types.h" COPYONLY)

add_library(mfph_ogg STATIC ${OGG}/src/bitwise.c ${OGG}/src/framing.c)
target_include_directories(mfph_ogg PUBLIC "${OGG}/include" "${MF_OGG_GENERATED}")

add_library(mfph_vorbis STATIC
	${VORBIS}/lib/mdct.c ${VORBIS}/lib/smallft.c ${VORBIS}/lib/block.c ${VORBIS}/lib/envelope.c ${VORBIS}/lib/window.c
	${VORBIS}/lib/lsp.c ${VORBIS}/lib/lpc.c ${VORBIS}/lib/analysis.c ${VORBIS}/lib/synthesis.c ${VORBIS}/lib/psy.c
	${VORBIS}/lib/info.c ${VORBIS}/lib/floor1.c ${VORBIS}/lib/floor0.c ${VORBIS}/lib/res0.c ${VORBIS}/lib/mapping0.c
	${VORBIS}/lib/registry.c ${VORBIS}/lib/codebook.c ${VORBIS}/lib/sharedbook.c ${VORBIS}/lib/lookup.c
	${VORBIS}/lib/bitrate.c ${VORBIS}/lib/vorbisenc.c ${VORBIS}/lib/vorbisfile.c)
target_include_directories(mfph_vorbis PUBLIC "${VORBIS}/include" PRIVATE "${VORBIS}/lib")
target_link_libraries(mfph_vorbis PUBLIC mfph_ogg)
if(NOT WIN32)
	target_link_libraries(mfph_vorbis PUBLIC m)
endif()

foreach(lib mfph_ogg mfph_vorbis)
	if(MSVC)
		target_compile_options(${lib} PRIVATE /w)
		target_compile_definitions(${lib} PRIVATE _CRT_SECURE_NO_WARNINGS)
	else()
		target_compile_options(${lib} PRIVATE -w)
	endif()
endforeach()

if(MF_PLUGINHOST_VST3)
	include(vst3.cmake)
endif()

# RtMidi, static, one API per OS. Linux without the ALSA headers still builds: RtMidi falls back to its dummy API and
# `hello` leaves CapabilityMidi out (as it does when ALSA fails to initialise at run time).
if(MF_PLUGINHOST_MIDI)
	set(RTMIDI "${CMAKE_CURRENT_SOURCE_DIR}/external/rtmidi")
	add_library(mfph_rtmidi STATIC ${RTMIDI}/RtMidi.cpp)
	target_include_directories(mfph_rtmidi SYSTEM PUBLIC "${RTMIDI}")
	target_compile_features(mfph_rtmidi PRIVATE cxx_std_11)
	if(APPLE)
		target_compile_definitions(mfph_rtmidi PRIVATE __MACOSX_CORE__)
		target_link_libraries(mfph_rtmidi PUBLIC "-framework CoreMIDI" "-framework CoreAudio" "-framework CoreFoundation")
	elseif(WIN32)
		target_compile_definitions(mfph_rtmidi PRIVATE __WINDOWS_MM__)
		target_link_libraries(mfph_rtmidi PUBLIC winmm)
	else()
		find_package(ALSA QUIET)
		find_package(Threads REQUIRED)
		target_link_libraries(mfph_rtmidi PUBLIC Threads::Threads)
		if(ALSA_FOUND)
			target_compile_definitions(mfph_rtmidi PRIVATE __LINUX_ALSA__)
			target_link_libraries(mfph_rtmidi PUBLIC ALSA::ALSA)
		else()
			message(WARNING "mfplughost: ALSA headers not found (libasound2-dev); MIDI input disabled in this build")
		endif()
	endif()
	if(MSVC)
		target_compile_options(mfph_rtmidi PRIVATE /w)
	else()
		target_compile_options(mfph_rtmidi PRIVATE -w)
	endif()
endif()

# Everything but main(): the executable and the tests link it.
add_library(mfph_core STATIC
	src/wav.hpp src/wav.cpp
	src/encode.hpp src/encode.cpp
	src/protocol.hpp src/protocol.cpp
	src/transport.hpp src/transport.cpp
	src/server.hpp src/server.cpp
	src/shm.hpp src/shm.cpp
	src/mainthread.hpp
	src/midi.hpp src/midi.cpp
	src/plugins.hpp
	src/version.hpp)
if(MF_PLUGINHOST_VST3)
	target_sources(mfph_core PRIVATE src/plugins.cpp)
	target_link_libraries(mfph_core PUBLIC mfph_vst3_hosting)
	target_compile_definitions(mfph_core PUBLIC MFPH_VST3=1)
else()
	target_sources(mfph_core PRIVATE src/plugins_none.cpp)
endif()
if(MF_PLUGINHOST_MIDI)
	target_link_libraries(mfph_core PUBLIC mfph_rtmidi)
	target_compile_definitions(mfph_core PUBLIC MFPH_MIDI=1)
endif()
if(APPLE)
	target_sources(mfph_core PRIVATE src/mainthread_mac.mm)
	set_source_files_properties(src/mainthread_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
	target_link_libraries(mfph_core PUBLIC "-framework Cocoa")
else()
	target_sources(mfph_core PRIVATE src/mainthread.cpp)
endif()
target_include_directories(mfph_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(mfph_core PUBLIC MFPH_VERSION="${PROJECT_VERSION}")
target_link_libraries(mfph_core PUBLIC mfph_vorbis)
if(WIN32)
	target_link_libraries(mfph_core PUBLIC ws2_32)
	target_compile_definitions(mfph_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()

add_executable(mfplughost src/main.cpp)
target_link_libraries(mfplughost PRIVATE mfph_core)
if(WIN32)
	target_link_libraries(mfplughost PRIVATE shell32)
elseif(NOT APPLE)
	target_link_options(mfplughost PRIVATE -static-libstdc++ -static-libgcc)
endif()

foreach(target mfph_core mfplughost)
	target_compile_features(${target} PRIVATE cxx_std_20)
	set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
	if(MSVC)
		target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus $<$<BOOL:${MF_NATIVES_WERROR}>:/WX>)
	else()
		target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wsign-conversion
			$<$<BOOL:${MF_NATIVES_WERROR}>:-Werror>)
	endif()
endforeach()

if(BUILD_TESTING)
	add_executable(mfplughost_tests tests/tests.cpp)
	target_link_libraries(mfplughost_tests PRIVATE mfph_core)
	target_compile_features(mfplughost_tests PRIVATE cxx_std_20)
	if(MF_PLUGINHOST_VST3)
		add_subdirectory(tests/plugins)
		add_dependencies(mfplughost_tests mf_test_plugins)
		target_compile_definitions(mfplughost_tests PRIVATE MFPH_TEST_PLUGINS="${MF_TEST_PLUGINS_BUNDLE}")
	endif()
	add_test(NAME mfplughost_tests COMMAND mfplughost_tests "$<TARGET_FILE:mfplughost>" "${CMAKE_CURRENT_BINARY_DIR}/test-output")
endif()
